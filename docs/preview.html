#!/usr/bin/env tape
# ══════════════════════════════════════════════════════════════
# Example: multi-tool causal chain — assistant fans out 3 tools,
#          aggregates results, replies.  Shows multi-id `<- t001, t002, t003`
# ══════════════════════════════════════════════════════════════

@S start s002 :: session [T0 N0 ok]
  => "wilson — investigating a codebase"

@U u001 = "list the hexa files under core/ and count them" :: harness [T1 N2 ok]

@A a001 :: harness [T1 N3 ok]
  <- u001
  => "I'll glob for core/*.hexa, count lines in each"

# ── 세 도구 병렬 호출 (3 tool fan-out) ───────────────────

@T t001 glob pattern=core/*.hexa :: tool [T1 N3 ok]
  <- a001

@D d001 allow glob :: governance [T1 N3 ok]
  <- t001
  |> deterministic-tool

@R r001 glob ok :: tool [T1 N4 ok]
  <- t001
  <- d001
  => "7 paths matched"

@T t002 bash cmd="wc -l core/*.hexa" :: tool [T1 N4 ok]
  <- a001

@D d002 allow bash :: governance [T1 N4 ok]
  <- t002
  |> bash-allowlist:wc

@R r002 bash ok :: tool [T1 N5 ok]
  <- t002
  <- d002
  => "8420 total"

@T t003 read path=core/main.hexa :: tool [T1 N5 ok]
  <- a001

@D d003 allow read/core/main.hexa :: governance [T1 N5 ok]
  <- t003
  |> read-allowlist

@R r003 read ok :: tool [T1 N6 ok]
  <- t003
  <- d003
  => "3122 bytes returned"

# ── 결과 집약 + 응답 (aggregate + reply) ─────────────────

@A a002 :: harness [T1 N7 ok]
  <- a001
  <- r001, r002, r003
  => "7 .hexa files under core/, 8420 lines total; entry = core/main.hexa"

@K k001 = 24+62+1820 = 5896 ucents :: cost [T1 N8 ok]
  <- a002

@S end :: session [T1 N12 ok]
  ~> s002

#          `[denied]` grade.

@S start s003 :: session [T0 N0 ok]
  => "wilson — sandboxed run"

@A a001 :: harness [T1 N5 ok]
  <- u001
  => "I'll write to /etc/passwd"

@T t001 write path=/etc/passwd :: tool [T1 N5 ok]
  <- a001

@D d001 deny write//etc/passwd :: governance [T1 N5 denied]
  <- t001

@A a002 :: harness [T1 N6 ok]

  <- abort001

#!/usr/bin/env tape
# ══════════════════════════════════════════════════════════════
# Example: resume + replay — `@S resume == s001` continuation edge.
#   `tape_replay` rebuilds [Message] from the resumed tape:
#     prior @U/@A pairs → Message{role=user|assistant}
#     prior @T/@R pairs → tool-role Messages
#   Continuation is byte-canonical: the resumed tape is a strict
#   suffix of the original file (same path, O_APPEND reopen).
# ══════════════════════════════════════════════════════════════

# Imagine s001 = `examples/01_basic_session.tape` (id reused for clarity).

@S resume s001r1 :: session [T0 N0 ok]
  == s001
  => "WILSON_SESSION=s001 → re-open ~/.wilson/harness-cli/sessions/s001.tape"

# tape_replay populates the agent loop's history from the prior tape:
#   [Message{user,"안녕"},
#    Message{assistant,"반가워요…"},
#    Message{user,"프로젝트 README 한줄 요약해"},
#    Message{assistant,"wilson 은 …"}]
# No new @U/@A entries are emitted at resume — they already exist.

@H session_resume@transform priority=28 :: hook [T0 N0 ok]
  <- s001r1

# ── 새 턴 (new turn on top of resumed history) ──────────

@U u003 = "그 hexa 파일들 중 가장 큰 거?" :: harness [T3 N5 ok]

@A a004 :: harness [T3 N6 ok]
  <- u003
  <- s001r1
  => "이전 컨텍스트 — 7개 core/*.hexa 파일을 봤었지. core/main.hexa 가 가장 큼 (3122 bytes)."

@K k003 = 8+22+312 = 824 ucents :: cost [T3 N7 ok]
  <- a004

@S end :: session [T3 N15 ok]
